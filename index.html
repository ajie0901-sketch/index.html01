const toText = document.getElementById('est-to').value.trim(); if (!fromText || !toText) { alert("請填寫或定位出發地與目的地！"); return; } const mapContainer = document.getElementById('map-container'); const mapFrame = document.getElementById('google-map-frame'); mapFrame.src = googleEmbedUrl; let routeKm = 10.0; // 預設預防性距離 let routeMin = 25; // 預設預防性時間 try { // 2. 利用地理編碼取得起終點座標來精算真實路徑 const resFrom = await fetch(`https://nominatim.openstreetmap.org/search?format=json&q=${encodeURIComponent(fromText)}`); const dataFrom = await resFrom.json(); const resTo = await fetch(`https://nominatim.openstreetmap.org/search?format=json&q=${encodeURIComponent(toText)}`); const dataTo = await resTo.json(); if (dataFrom.length > 0 && dataTo.length > 0) { const lat1 = parseFloat(dataFrom[0].lat); const lon1 = parseFloat(dataFrom[0].lon); const lat2 = parseFloat(dataTo[0].lat); const lon2 = parseFloat(dataTo[0].lon); // 使用 OSRM 免費公開導航引擎 API 取得跟 Google Maps 高度一致的實際車行距離與秒數 const osrmRes = await fetch(`https://router.project-osrm.org/route/v1/driving/${lon1},${lat1};${lon2},${lat2}?overview=false`); const osrmData = await osrmRes.json(); if (osrmData.code === "Ok" && osrmData.routes && osrmData.routes.length > 0) { const route = osrmData.routes[0]; routeKm = (route.distance / 1000).toFixed(1); // 公里 routeMin = Math.round(route.duration / 60); const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a)); routeKm = (R * c * 1.3).toFixed(1); routeMin = Math.round(routeKm * 2.4); } } } catch (e) { console.log("導航數據取得異常，使用估算值"); } // 3. 抓取設定分頁裡的動態計費參數 const baseFare = parseFloat(document.getElementById('rate-base')?.value) || 75; total += overKm * extraKmRate; } // 套用夜間加成 total *= nightMultiplier; // 5. 即時更新畫面上的價格與「同步後的導航距離與時間」 document.getElementById('result-price').innerText = `NT$ ${Math.round(total)}`; document.getElementById('result-info').innerText = `導航距離：${routeKm} 公里 | 預估行車時間：${routeMin} 分鐘`;
